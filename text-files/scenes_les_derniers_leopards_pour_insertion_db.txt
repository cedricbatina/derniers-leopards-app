START TRANSACTION;

-- 1) Recuperer le livre/projet
SET @book_slug := 'les-derniers-leopards';
-- SET @project_slug := 'les-derniers-leopards';

-- 2) Exemple d'insertion de scene dans le schema ancien
-- Remplacer les valeurs par les lignes du format pipe.
INSERT INTO scenes (
  chapter_id, scene_no, slug, title,
  pov_character_id, location_id,
  summary, conflict, turning_point, outcome, hook, indesign_style
)
SELECT
  c.id,
  1,
  'p1c1-s1-mbongi-1702',
  'Mbanza Kongo se releve',
  cp.id,
  l.id,
  'Kimpa Vita et les mampuya maa Kongo s installent a Mbanza Kongo pour guerir et rebâtir. La rumeur atteint Mutinu.',
  'La capitale est blessee ; la memoire doit trouver une bouche.',
  'Mutinu decide de venir.',
  'Le cadre du mbongi de memoire s ouvre.',
  'Le nzonzi prepare le retour vers le passe.',
  'scene-standard'
FROM chapters c
JOIN parts p ON p.id = c.part_id
JOIN books b ON b.id = p.book_id
LEFT JOIN characters cp ON cp.slug = 'mutinu'
LEFT JOIN locations l ON l.name = 'Mbanza Kongo'
WHERE b.slug = @book_slug
  AND p.slug = 'temple-et-rappel'
  AND c.slug = 'prologue-jumeaux'
ON DUPLICATE KEY UPDATE
  title = VALUES(title),
  summary = VALUES(summary),
  conflict = VALUES(conflict),
  turning_point = VALUES(turning_point),
  outcome = VALUES(outcome),
  hook = VALUES(hook),
  indesign_style = VALUES(indesign_style);

COMMIT;
